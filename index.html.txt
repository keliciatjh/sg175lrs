    let audioCtx = null;
    let audioMuted = false;

    function initAudioContext() {
      if (!audioCtx) {
        audioCtx = new (window.AudioContext || window.webkitAudioContext)();
      }
    }

    function toggleAudioMute() {
      audioMuted = !audioMuted;
      const btn = document.getElementById('audio-icon');
      if (audioMuted) {
        btn.className = 'fa-solid fa-volume-xmark text-red-400 text-xs';

    function playSynthSound(type) {
      if (audioMuted) return;
      initAudioContext();
      if (!audioCtx) return;

      const now = audioCtx.currentTime;
      const osc = audioCtx.createOscillator();
      const gain = audioCtx.createGain();

      osc.connect(gain);
      gain.connect(audioCtx.destination);

      switch(type) {
        case 'night_start':
          // Low pitch ambient drop
          osc.type = 'sine';
          osc.frequency.setValueAtTime(240, now);
          osc.frequency.exponentialRampToValueAtTime(80, now + 1.2);
          gain.gain.setValueAtTime(0.3, now);
          gain.gain.exponentialRampToValueAtTime(0.01, now + 1.2);
          osc.start(now);
          osc.stop(now + 1.2);
          break;

        case 'wolf_howl':
          // Modulated sawtooth wolf howl
          osc.type = 'sawtooth';
          osc.frequency.setValueAtTime(280, now);
          osc.frequency.linearRampToValueAtTime(520, now + 0.4);
          osc.frequency.linearRampToValueAtTime(360, now + 1.1);
          gain.gain.setValueAtTime(0.18, now);
          gain.gain.exponentialRampToValueAtTime(0.01, now + 1.1);
          osc.start(now);
          osc.stop(now + 1.1);
          break;

        case 'day_bell':
          // High triangle chime bell
          osc.type = 'triangle';
          osc.frequency.setValueAtTime(880, now);
          gain.gain.setValueAtTime(0.4, now);
          gain.gain.exponentialRampToValueAtTime(0.001, now + 1.5);
          osc.start(now);
          osc.stop(now + 1.5);
          break;

        case 'seer_check':
          // Dual harmonic tone
          osc.type = 'sine';
          osc.frequency.setValueAtTime(440, now);
          osc.frequency.setValueAtTime(660, now + 0.2);
          gain.gain.setValueAtTime(0.2, now);
          gain.gain.exponentialRampToValueAtTime(0.01, now + 0.5);
          osc.start(now);
          osc.stop(now + 0.5);
          break;

          gain.gain.exponentialRampToValueAtTime(0.01, now + 0.4);
          osc.start(now);
          osc.stop(now + 0.4);
          break;

        case 'timer_tick':
          // Short percussion pulse
          osc.type = 'square';
          osc.frequency.setValueAtTime(1000, now);
          gain.gain.setValueAtTime(0.1, now);
          gain.gain.exponentialRampToValueAtTime(0.001, now + 0.05);

    function applyPreset() {
      const selector = document.getElementById('preset-selector');
      if (selector) gameState.presetKey = selector.value;
      const config = PRESETS[gameState.presetKey];

      document.getElementById('preset-total-seats').textContent = config.total;
      document.getElementById('preset-wolf-count').textContent = config.roles.filter(r => r.includes('wolf')).length;
      document.getElementById('preset-god-count').textContent = config.roles.filter(r => ['seer','witch','hunter','guard','idiot'].includes(r)).length;

      gameState.players = [];
      for (let i = 1; i <= config.total; i++) {
        gameState.players.push({
          seat: i,
          name: `Player ${i}`,
          role: config.roles[i - 1] || 'villager',
          isAlive: true,
          isSheriff: false
        });
      }

      renderSeatsGrid();
      updateNightResolverDropdowns();
      updateVotingMatrixInputs();
      saveCurrentStateToStorage();
    }

    function updatePlayerRole(seat, val) {

    function toggleAlive(seat) {
      const p = gameState.players.find(x => x.seat === seat);
      if (p) {

        renderSeatsGrid();

      gameState.players.forEach(p => {

    function shuffleRoles() {
      const config = PRESETS[gameState.presetKey];
      const shuffled = [...config.roles].sort(() => Math.random() - 0.5);

      gameState.players.forEach((p, idx) => {
        p.role = shuffled[idx];
      });

      renderSeatsGrid();
      playSynthSound('day_bell');
      saveCurrentStateToStorage();

    function nextPhase() {
      if (gameState.currentPhaseIndex < PHASES.length - 1) {
        gameState.currentPhaseIndex++;
      } else {
        gameState.currentPhaseIndex = 1;
        gameState.dayCount++;
      }

      if (gameState.currentPhaseIndex === 1) playSynthSound('night_start');
      if (gameState.currentPhaseIndex === 4) playSynthSound('day_bell');

      updatePhaseUI();
      saveCurrentStateToStorage();
    }

    function previousPhase() {
      if (gameState.currentPhaseIndex > 0) {
        gameState.currentPhaseIndex--;
        updatePhaseUI();
        saveCurrentStateToStorage();
      }
    }

    let timerInterval = null;
    function startTimer(seconds) {
      clearInterval(timerInterval);
      let remaining = seconds;

      const circle = document.getElementById('timer-circle');
      const display = document.getElementById('timer-display');
      const circumference = 2 * Math.PI * 18;

      circle.style.strokeDasharray = `${circumference}`;
      playSynthSound('day_bell');

      timerInterval = setInterval(() => {
        remaining--;
        display.textContent = `${remaining}s`;

        const offset = circumference - (remaining / seconds) * circumference;
        circle.style.strokeDashoffset = offset;

        if (remaining <= 10 && remaining > 0) {
          playSynthSound('timer_tick');
        }

        if (remaining <= 0) {
          clearInterval(timerInterval);
          playSynthSound('day_bell');
          display.textContent = 'END';
        }
      }, 1000);
    }

    function stopTimer() {
      clearInterval(timerInterval);
      document.getElementById('timer-display').textContent = 'STOP';
    }

    function updateNightResolverDropdowns() {
      const wolfSel = document.getElementById('night-target-wolf');

    function resolveNightActions() {
      const wolfTarget = document.getElementById('night-target-wolf').value;
      const guardTarget = document.getElementById('night-target-guard').value;
      const witchSave = document.getElementById('night-target-witch-save').value === 'yes';
      const witchPoisonTarget = document.getElementById('night-target-witch-poison').value;

      let deaths = [];
      let notes = [];

      let wolfKilledSeat = wolfTarget !== 'none' ? parseInt(wolfTarget) : null;
      let guardProtectedSeat = guardTarget !== 'none' ? parseInt(guardTarget) : null;
      let poisonKilledSeat = witchPoisonTarget !== 'none' ? parseInt(witchPoisonTarget) : null;

      if (wolfKilledSeat) {
        const isGuarded = (wolfKilledSeat === guardProtectedSeat);
        const isSaved = witchSave;

        if (isGuarded && isSaved) {
          deaths.push(wolfKilledSeat);
          notes.push(`Seat #${wolfKilledSeat} died due to DOUBLE PROTECTION (Tong Shou Rule).`);
        } else if (isGuarded || isSaved) {
          notes.push(`Seat #${wolfKilledSeat} was attacked but SAVED by ${isGuarded ? 'Guard' : 'Witch Potion'}.`);
        } else {
          deaths.push(wolfKilledSeat);
          notes.push(`Seat #${wolfKilledSeat} was killed by Werewolves.`);
        }
      }

      if (poisonKilledSeat) {
        deaths.push(poisonKilledSeat);
        notes.push(`Seat #${poisonKilledSeat} was eliminated by Witch Poison.`);
      }

      const box = document.getElementById('night-resolution-box');
      box.classList.remove('hidden');

      if (deaths.length === 0) {
        box.innerHTML = `<span class="text-emerald-400 font-bold"><i class="fa-solid fa-sun"></i> PEACEFUL NIGHT:</span> No players died tonight.`;
      } else {
        const uniqueDeaths = [...new Set(deaths)];
        box.innerHTML = `
          <div class="text-red-400 font-bold mb-1"><i class="fa-solid fa-skull"></i> NIGHT ELIMINATIONS:</div>
          <div class="text-white">Players Died: ${uniqueDeaths.map(d => `<strong class="text-amber-300">Seat #${d}</strong>`).join(', ')}</div>
          <div class="text-[10px] text-gray-400 mt-1">${notes.join('<br>')}</div>
        `;

        uniqueDeaths.forEach(d => {
          const p = gameState.players.find(x => x.seat === d);
          if (p) p.isAlive = false;
        });
        renderSeatsGrid();
      }
      saveCurrentStateToStorage();
    }

    function tallyVotes() {
      const votes = {};
      const inputs = document.querySelectorAll('.vote-input');

      inputs.forEach(input => {
        const voterSeat = parseInt(input.dataset.voter);
        const targetSeat = input.value;

        if (targetSeat !== 'abstain') {
          const voter = gameState.players.find(p => p.seat === voterSeat);
          const weight = (voter && voter.isSheriff) ? 1.5 : 1.0;
          votes[targetSeat] = (votes[targetSeat] || 0) + weight;
        }
      });

      const resultBox = document.getElementById('vote-result-box');
      resultBox.classList.remove('hidden');

      if (Object.keys(votes).length === 0) {
        resultBox.textContent = "All players abstained from voting.";
        return;
      }

      let maxVotes = 0;
      let executedSeat = null;
      let tie = false;

      for (const [target, count] of Object.entries(votes)) {
        if (count > maxVotes) {
          maxVotes = count;
          executedSeat = target;
          tie = false;
        } else if (count === maxVotes) {
          tie = true;
        }
      }

      if (tie) {
        resultBox.innerHTML = `<span class="text-amber-400 font-bold">TIE VOTE DETECTED (${maxVotes} pts).</span> A runoff speech and re-vote is required.`;
      } else {
        const executedPlayer = gameState.players.find(p => p.seat === parseInt(executedSeat));
        resultBox.innerHTML = `
          <div class="font-bold text-red-400">EXECUTED BY DAY VOTE:</div>
          <div>Seat #${executedSeat} (${executedPlayer ? executedPlayer.name : ''}) with <strong>${maxVotes}</strong> total votes (Sheriff = 1.5x).</div>
        `;
      }
    }
